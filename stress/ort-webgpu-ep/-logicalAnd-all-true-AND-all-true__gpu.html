function filledFloat32(shape, value) {
  const size = Array.isArray(shape) ? shape.reduce((a, b) => a * b, 1) : shape;

  const size = Array.isArray(shape) ? shape.reduce((a, b) => a * b, 1) : shape;
  const arr = new Float32Array(size);

}
